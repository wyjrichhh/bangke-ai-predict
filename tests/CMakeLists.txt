cmake_minimum_required(VERSION 3.21)
project(AIPredictTraceTests LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Standalone focused tests reuse a compatible Rime build and its dependencies.
set(RIME_ROOT "" CACHE PATH "Compatible librime source and dependency prefix")
set(RIME_BUILD_ROOT "${RIME_ROOT}/build" CACHE PATH "Compatible configured librime build")
find_library(AI_TEST_RIME_LIBRARY NAMES rime PATHS "${RIME_BUILD_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
find_library(AI_TEST_GLOG_LIBRARY NAMES glog PATHS "${RIME_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
find_path(AI_TEST_BOOST_INCLUDE NAMES boost/signals2/connection.hpp HINTS "$ENV{BOOST_ROOT}" /opt/homebrew/include REQUIRED)
include_directories("${RIME_ROOT}/src" "${RIME_BUILD_ROOT}/src" "${RIME_ROOT}/include" "${AI_TEST_BOOST_INCLUDE}")
add_compile_definitions(GLOG_USE_GLOG_EXPORT GLOG_STATIC_DEFINE)
if(WIN32)
  add_compile_definitions(RIME_IMPORTS)
endif()
set(rime_library "${AI_TEST_RIME_LIBRARY}")
add_subdirectory(.. plugin EXCLUDE_FROM_ALL)
add_executable(predict_trace_test predict_trace_test.cc $<TARGET_OBJECTS:rime-ai-predict-objs>)
target_include_directories(predict_trace_test PRIVATE ../src)
target_include_directories(predict_trace_test PRIVATE "${CTRANSLATE2_ROOT}/include")
target_link_libraries(predict_trace_test PRIVATE ${plugin_deps} "${AI_TEST_GLOG_LIBRARY}")
if(WIN32)
  # Supply only non-exported helpers; exported Translation methods stay in the DLL.
  target_sources(predict_trace_test PRIVATE rime_windows_helpers.cc)
  target_link_libraries(predict_trace_test PRIVATE dbghelp)
endif()
if(MSVC)
  target_compile_options(predict_trace_test PRIVATE /W4 /utf-8)
  target_compile_options(rime-ai-predict-objs PRIVATE /utf-8)
  set_property(TARGET predict_trace_test rime-ai-predict-objs PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_compile_options(predict_trace_test PRIVATE -Wall -Wextra)
endif()
add_test(NAME predict_trace COMMAND predict_trace_test)
set(BANGKE_AI_TEST_MODEL_DIR "" CACHE PATH "Optional existing real CT2 model for an integration load and prediction")
if(BANGKE_AI_TEST_MODEL_DIR)
  add_test(NAME predict_trace_model COMMAND predict_trace_test "${BANGKE_AI_TEST_MODEL_DIR}")
endif()
